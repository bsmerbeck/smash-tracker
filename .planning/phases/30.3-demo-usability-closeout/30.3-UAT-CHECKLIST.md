  ```sh
  ENV_FILE="apps/web/.env.production"
  WEB_PREFLIGHT_STATUS=0
  if [ ! -f "$ENV_FILE" ]; then

    for VAR in VITE_FIREBASE_API_KEY VITE_FIREBASE_AUTH_DOMAIN VITE_FIREBASE_PROJECT_ID VITE_FIREBASE_APP_ID; do
  ```
